=== grammar ===

Program -> Function*

Function -> func identifier Args Body

Args -> par_open (identifier (comma identifier)*)? par_close

Body -> brace_open Statement* brace_close

Statement -> Decl end
           | Assign end
           | Call end
           | Return end
           | If
           | While
           | Body

Decl -> var identifier is Expr
Assign -> identifier Index* is Expr
Call -> identifier par_open ExprList? par_close
Return -> return Expr

Index -> bracket_open Expr bracket_close
ExprList -> Expr (comma Expr)*

If -> if Expr Body Else?
Else -> else If | else Body
While -> while Expr Body

Expr
    : AndExpr
    | Expr or AndExpr

AndExpr
    : EqualityExpr
    | AndExpr and EqualityExpr

EqualityExpr
    : RelationExpr
    | EqualityExpr eq RelationExpr
    | EqualityExpr neq RelationExpr

RelationExpr
    : AdditiveExpr
    | RelationExpr lt AdditiveExpr
    | RelationExpr gt AdditiveExpr
    | RelationExpr lte AdditiveExpr
    | RelationExpr gte AdditiveExpr

AdditiveExpr
    : multiplicativeExpr
    | AdditiveExpr add MultiplicativeExpr
    | AdditiveExpr sub MultiplicativeExpr

MultiplicativeExpr
    : ExponentialExpr
    | MultiplicativeExpr mul ExponentialExpr
    | MultiplicativeExpr div ExponentialExpr

ExponentialExpr
    : UnaryExpr
    | ExponentialExpr pow UnaryExpr

UnaryExpr
    : UnitExpr
    | sub UnitExpr
    | not UnitExpr
    | mul UnitExpr
    | addr UnitExpr

UnitExpr
    : par_open Expr par_close
    | Call
    | integer
    | character
    | string
    | identifier

=== ast ===

program = function*

function = identifier argument* body

identifier = string

argument = identifier

body = statement*

statement 
    = declaration
    | assignment
    | call
    | return
    | if
    | while

if = expression body body?

while = expression body

declaration = identifier expression

assignment = identifier expression* expression

call = identifier expression*

return = expression?

expression 
    = binaryoperator expression expression
    | unaryoperator expression
    | integer
    | boolean
    | character
    | identifier
    | call

binaryoperator 
    = +
    | - 
    | *
    | /
    | ^
    | &
    | |
    | ==
    | !=
    | <
    | >
    | >=
    | =<

unaryoperator
    = -
    | !
    | &
    | *

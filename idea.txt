    a[at] = with;
}

=== grammar ===

Program -> Function*

Function -> func identifier Args (arrow Type)? Body

Args -> par_open (Arg (comma Arg)*)? par_close
Arg -> identifier colon Type

Body -> brace_open Statement* brace_close

Statement ->
    | Decl end
    | Assign end
    | Call end
    | Return end
    | If
    | While
    | Body

Decl -> var identifier (colon Type) is Expr
Assign -> identifier (bracket_open Expr bracket_close)* is Expr
Call -> identifier par_open (Expr (comma Expr)*)? par_close
Return -> return Expr

If -> if Expr Body Else?
Else ->
    | else If
    | else Body
While -> while Expr Body

Expr ->
    | AndExpr
    | Expr or AndExpr

AndExpr ->
    | EqualityExpr
    | AndExpr and EqualityExpr

EqualityExpr ->
    | RelationExpr
    | EqualityExpr eq RelationExpr
    | EqualityExpr neq RelationExpr

RelationExpr ->
    | AdditiveExpr
    | RelationExpr lt AdditiveExpr
    | RelationExpr gt AdditiveExpr
    | RelationExpr lte AdditiveExpr
    | RelationExpr gte AdditiveExpr

AdditiveExpr ->
    | multiplicativeExpr
    | AdditiveExpr plus MultiplicativeExpr
    | AdditiveExpr minus MultiplicativeExpr

MultiplicativeExpr ->
    | ExponentialExpr
    | MultiplicativeExpr star ExponentialExpr
    | MultiplicativeExpr slash ExponentialExpr

ExponentialExpr ->
    | UnaryExpr
    | ExponentialExpr pow UnaryExpr

UnaryExpr ->
    | UnitExpr
    | minus UnitExpr
    | not UnitExpr
    | star UnitExpr
    | and UnitExpr

UnitExpr ->
    | par_open Expr par_close
    | Call
    | integer
    | boolean
    | character
    | string
    | identifier

Type ->
    | bracket_open Type bracket_close
    | and Type
    | integer_type
    | boolean_type
    | char_type

=== ast ===

program = function*

function = identifier argument* type? body

identifier = string

argument = identifier Type

body = statement*

statement 
    = declaration
    | assignment
    | call
    | return
    | if
    | while

if = expression body body

while = expression body

declaration = identifier type? expression

assignment = identifier expression* expression

call = identifier expression*

return = expression?

expression 
    = binaryoperator expression expression
    | unaryoperator expression
    | integer
    | boolean
    | character
    | identifier
    | call

binaryoperator 
    = +
    | - 
    | *
    | /
    | ^
    | &
    | |
    | ==
    | !=
    | <
    | >
    | >=
    | =<

unaryoperator
    = -
    | !
    | &
    | *

type
    = array
    | pointer
    | integer
    | boolean
    | character

array = type
pointer = type
